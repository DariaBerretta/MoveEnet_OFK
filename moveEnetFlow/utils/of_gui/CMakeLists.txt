cmake_minimum_required(VERSION 3.16)
project(of_sanity_gui_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(YARP COMPONENTS os sig cv REQUIRED)
find_package(OpenCV REQUIRED)
find_package(event-driven REQUIRED)
find_package(hpe-core REQUIRED)
add_executable(of_sanity_gui_engine engine.cpp)
target_include_directories(of_sanity_gui_engine PRIVATE ${YARP_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(of_sanity_gui_engine PRIVATE ${YARP_LIBRARIES} ${OpenCV_LIBS} ev::event-driven hpe-core::hpe-core)
